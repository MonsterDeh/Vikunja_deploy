# /etc/nginx/sites-available/vikunja
# Reverse proxy config برای Vikunja (نسخه‌ی جدید و یکپارچه vikunja/vikunja)
#
# نکته‌ی مهم: در این نسخه دیگه frontend و api دو container جدا نیستن؛
# همه‌چیز (هم رابط کاربری هم API) از یک سرویس به اسم "vikunja" روی
# پورت داخلی 3456 سرو می‌شه. پس دیگه نیازی به دو تا location جدا
# (یکی برای frontend یکی برای api) نیست.

server {
    listen 80;
    listen [::]:80;
    server_name myapp.local;

    # حداکثر حجم فایل آپلودی (attachments روی task ها)
    client_max_body_size 20M;

    location / {
        proxy_pass http://vikunja:3456;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # برای WebSocket / اتصال‌های طولانی (در صورت نیاز آینده)
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }

    # ---- Error handling ----
    error_page 502 503 504 /50x.html;
    location = /50x.html {
        root /usr/share/nginx/html;
    }

    access_log /var/log/nginx/vikunja_access.log;
    error_log  /var/log/nginx/vikunja_error.log;
}
