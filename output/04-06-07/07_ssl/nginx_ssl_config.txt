# /etc/nginx/sites-available/vikunja
# نسخه‌ی نهایی با SSL/TLS برای Vikunja (image یکپارچه vikunja/vikunja)

# ---- 1) HTTP -> HTTPS redirect ----
server {
    listen 80;
    listen [::]:80;
    server_name myapp.local;

    return 301 https://$host$request_uri;
}

# ---- 2) HTTPS server ----
server {
    listen 443 ssl;
    listen [::]:443 ssl;
    server_name myapp.local;

    client_max_body_size 20M;

    # ---- SSL certificate ----
    ssl_certificate     /etc/nginx/ssl/myapp.local.crt;
    ssl_certificate_key /etc/nginx/ssl/myapp.local.key;

    # ---- SSL protocols & ciphers (Security best practices) ----
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5:!3DES;
    ssl_prefer_server_ciphers on;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;

    location / {
        proxy_pass http://vikunja:3456;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }

    error_page 502 503 504 /50x.html;
    location = /50x.html {
        root /usr/share/nginx/html;
    }

    access_log /var/log/nginx/vikunja_ssl_access.log;
    error_log  /var/log/nginx/vikunja_ssl_error.log;
}
